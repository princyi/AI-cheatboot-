@app.route(, methods=[GET, POST])
def home()

    answer = 
    error = 
    question = 

    if request.method == POST

        question = request.form.get(question, ).strip()

        if question

            start_time = time.perf_counter()
            success = False
            failure_type = None

            try

                response = requests.post(
                    OLLAMA_URL,
                    json={
                        model MODEL_NAME,
                        prompt question,
                        stream False
                    },
                    timeout=180
                )

                response.raise_for_status()

                answer = response.json()[response]
                success = True

            except (requests.RequestException, ValueError, KeyError) as exc

                error = str(exc)
                failure_type = type(exc).__name__

            finally

                response_time = time.perf_counter() - start_time

                # Log chatbot interaction in MLflow
                try

                    mlflow.set_experiment(MLFLOW_EXPERIMENT)

                    with mlflow.start_run()

                        mlflow.log_param(model, MODEL_NAME)
                        mlflow.log_param(framework, Ollama)
                        mlflow.log_param(question_length, len(question))

                        mlflow.log_metric(
                            response_time_seconds,
                            response_time
                        )

                        mlflow.log_metric(
                            success,
                            int(success)
                        )

                        mlflow.set_tag(
                            application,
                            LLM Chatbot
                        )

                        mlflow.set_tag(
                            status,
                            success if success else failure
                        )

                        mlflow.set_tag(
                            failure_type,
                            failure_type
                        )

                        # Conversation content is stored as MLflow artifacts
                        mlflow.log_text(
                            question,
                            conversationuser_question.txt
                        )

                        mlflow.log_text(
                            answer if success else error,
                            conversationai_response.txt
                        )

                except Exception

                    app.logger.exception(
                        MLflow logging failed
                    )

    return render_template_string(
        HTML,
        question=question,
        answer=answer,
        error=error
    )
